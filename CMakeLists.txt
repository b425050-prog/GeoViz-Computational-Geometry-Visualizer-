cmake_minimum_required(VERSION 3.16)
project(GeoViz LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Engine and verification tools need neither graphics nor a display.
add_library(geoviz_engine src/geometry.cpp src/algorithms.cpp src/io.cpp src/samples.cpp src/security_samples.cpp)
target_include_directories(geoviz_engine PUBLIC include)
add_executable(geoviz_tests tests/tests.cpp)
target_link_libraries(geoviz_tests PRIVATE geoviz_engine)
add_executable(geoviz_showcase tools/export_showcase.cpp)
target_link_libraries(geoviz_showcase PRIVATE geoviz_engine)
add_executable(geoviz_samples tools/export_samples.cpp)
target_link_libraries(geoviz_samples PRIVATE geoviz_engine)
enable_testing()
add_test(NAME engine_checks COMMAND geoviz_tests "${CMAKE_CURRENT_SOURCE_DIR}"
         "${CMAKE_CURRENT_BINARY_DIR}/test-output")
foreach(target geoviz_engine geoviz_tests geoviz_showcase geoviz_samples)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(WIN32 AND MINGW AND NOT target STREQUAL "geoviz_engine")
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()

# Disable the application for CI or engine development without raylib.
option(GEOVIZ_BUILD_APP "Build the raylib desktop application" ON)
if(GEOVIZ_BUILD_APP)
    set(RAYLIB_ROOT "C:/raylib/w64devkit" CACHE PATH "Matching raylib installation")
    find_path(RAYLIB_INCLUDE raylib.h HINTS "${RAYLIB_ROOT}/include" REQUIRED)
    find_library(RAYLIB_LIBRARY NAMES raylib HINTS "${RAYLIB_ROOT}/lib" REQUIRED)
    add_executable(geoviz src/main.cpp)
    target_include_directories(geoviz PRIVATE "${RAYLIB_INCLUDE}")
    target_link_libraries(geoviz PRIVATE geoviz_engine "${RAYLIB_LIBRARY}")
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(geoviz PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(WIN32)
        target_link_libraries(geoviz PRIVATE opengl32 gdi32 winmm)
        enable_language(RC)
        target_sources(geoviz PRIVATE assets/geoviz.rc)
        set_source_files_properties(assets/geoviz.rc PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}")
        if(MINGW)
            target_link_options(geoviz PRIVATE -static)
        endif()
    endif()
endif()
