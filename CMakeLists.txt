cmake_minimum_required(VERSION 3.21)
project(GeoViz VERSION 2.0.0 LANGUAGES CXX)
option(GEOVIZ_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)
option(GEOVIZ_COVERAGE "Enable GCC coverage instrumentation" OFF)
if(GEOVIZ_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(GEOVIZ_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE GEOVIZ_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT GEOVIZ_REVISION)
  set(GEOVIZ_REVISION "unversioned-source")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Engine and verification tools need neither graphics nor a display.
add_library(geoviz_engine src/geometry.cpp src/algorithms.cpp src/io.cpp src/csv.cpp src/security.cpp src/security_io.cpp src/samples.cpp src/security_samples.cpp)
target_include_directories(geoviz_engine PUBLIC include)
target_compile_definitions(geoviz_engine PUBLIC GEOVIZ_SOURCE_REVISION="${GEOVIZ_REVISION}")
add_executable(geoviz_tests tests/tests.cpp)
target_link_libraries(geoviz_tests PRIVATE geoviz_engine)
add_executable(geoviz_showcase tools/export_showcase.cpp)
target_link_libraries(geoviz_showcase PRIVATE geoviz_engine)
add_executable(geoviz_samples tools/export_samples.cpp)
target_link_libraries(geoviz_samples PRIVATE geoviz_engine)
add_executable(geoviz_regressions tests/regressions.cpp)
target_link_libraries(geoviz_regressions PRIVATE geoviz_engine)
add_executable(geoviz_benchmark tools/benchmark.cpp)
target_link_libraries(geoviz_benchmark PRIVATE geoviz_engine)
enable_testing()
foreach(concern geometry routing clustering persistence security transitions)
    add_test(NAME ${concern}_regressions COMMAND geoviz_regressions ${concern} "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/regression-output/${concern}")
    set_tests_properties(${concern}_regressions PROPERTIES TIMEOUT 90)
endforeach()
add_test(NAME engine_checks COMMAND geoviz_tests "${CMAKE_CURRENT_SOURCE_DIR}"
         "${CMAKE_CURRENT_BINARY_DIR}/test-output")
foreach(target geoviz_engine geoviz_tests geoviz_regressions geoviz_showcase geoviz_samples geoviz_benchmark)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(WIN32 AND MINGW AND NOT target STREQUAL "geoviz_engine")
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()

# Disable the application for CI or engine development without raylib.
option(GEOVIZ_BUILD_APP "Build the raylib desktop application" ON)
if(GEOVIZ_BUILD_APP)
    option(GEOVIZ_FETCH_RAYLIB "Build the pinned raylib 5.5 source" OFF)
    if(GEOVIZ_FETCH_RAYLIB)
        include(FetchContent)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
        FetchContent_MakeAvailable(raylib)
        set(GEOVIZ_RAYLIB raylib)
    else()
        set(RAYLIB_ROOT "C:/raylib/w64devkit" CACHE PATH "Matching raylib 5.5 installation")
        find_path(RAYLIB_INCLUDE raylib.h HINTS "${RAYLIB_ROOT}/include" REQUIRED)
        find_library(RAYLIB_LIBRARY NAMES raylib HINTS "${RAYLIB_ROOT}/lib" REQUIRED)
        set(GEOVIZ_RAYLIB "${RAYLIB_LIBRARY}")
    endif()
    add_executable(geoviz src/main.cpp src/file_dialogs.cpp)
    if(RAYLIB_INCLUDE)
      target_include_directories(geoviz PRIVATE "${RAYLIB_INCLUDE}")
    endif()
    target_link_libraries(geoviz PRIVATE geoviz_engine ${GEOVIZ_RAYLIB})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(geoviz PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_custom_target(run COMMAND geoviz --root "${CMAKE_CURRENT_SOURCE_DIR}" DEPENDS geoviz WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    install(TARGETS geoviz RUNTIME DESTINATION .)
    install(DIRECTORY assets data docs DESTINATION .)
    install(FILES README.md LICENSE DESTINATION .)
    if(WIN32)
        target_link_libraries(geoviz PRIVATE opengl32 gdi32 winmm comdlg32)
        enable_language(RC)
        target_sources(geoviz PRIVATE assets/geoviz.rc)
        set_source_files_properties(assets/geoviz.rc PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}" OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/geoviz.ico")
        if(MINGW)
            target_link_options(geoviz PRIVATE -static)
        endif()
    endif()
endif()

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME structured_evidence COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_evidence.py" "${CMAKE_CURRENT_BINARY_DIR}/test-output")
  set_tests_properties(structured_evidence PROPERTIES DEPENDS engine_checks TIMEOUT 30)
  add_custom_target(source_checks COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_release.py" --source WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
