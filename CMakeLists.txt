cmake_minimum_required(VERSION 3.21)
project(GeoViz VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GEOVIZ_BUILD_APP "Build the raylib desktop application" ON)
option(GEOVIZ_FETCH_RAYLIB "Build pinned raylib 5.5 instead of an installed copy" OFF)
option(GEOVIZ_SANITIZERS "Enable address and undefined-behaviour checks on Linux" OFF)
if(GEOVIZ_SANITIZERS AND NOT WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
    OUTPUT_VARIABLE GEOVIZ_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT GEOVIZ_REVISION)
    set(GEOVIZ_REVISION "unversioned-source")
endif()
add_library(geoviz_engine src/geometry.cpp src/algorithms.cpp src/io.cpp src/csv.cpp
    src/security.cpp src/security_io.cpp src/samples.cpp src/security_samples.cpp)
target_include_directories(geoviz_engine PUBLIC include)
target_compile_definitions(geoviz_engine PUBLIC GEOVIZ_SOURCE_REVISION="${GEOVIZ_REVISION}")
set(checked_targets geoviz_engine)

include(CTest)
if(BUILD_TESTING)
    foreach(name tests regressions)
        add_executable(geoviz_${name} tests/${name}.cpp)
        target_link_libraries(geoviz_${name} PRIVATE geoviz_engine)
        list(APPEND checked_targets geoviz_${name})
    endforeach()
    foreach(concern geometry routing clustering persistence security transitions)
        add_test(NAME ${concern}_regressions COMMAND geoviz_regressions ${concern}
            "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/regression-output/${concern}")
        set_tests_properties(${concern}_regressions PROPERTIES TIMEOUT 90)
    endforeach()
    add_test(NAME engine_checks COMMAND geoviz_tests "${PROJECT_SOURCE_DIR}"
        "${PROJECT_BINARY_DIR}/test-output")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME structured_evidence COMMAND "${Python3_EXECUTABLE}"
            "${PROJECT_SOURCE_DIR}/tests/evidence_checks.py" "${PROJECT_BINARY_DIR}/test-output")
        set_tests_properties(structured_evidence PROPERTIES DEPENDS engine_checks TIMEOUT 30)
    endif()
endif()

if(GEOVIZ_BUILD_APP)
    if(GEOVIZ_FETCH_RAYLIB)
        include(FetchContent)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git
            GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
        FetchContent_MakeAvailable(raylib)
        set(GEOVIZ_RAYLIB raylib)
    else()
        set(RAYLIB_ROOT "C:/raylib/w64devkit" CACHE PATH "Matching raylib 5.5 installation")
        find_path(RAYLIB_INCLUDE raylib.h HINTS "${RAYLIB_ROOT}/include" REQUIRED)
        find_library(GEOVIZ_RAYLIB NAMES raylib HINTS "${RAYLIB_ROOT}/lib" REQUIRED)
    endif()
    add_executable(geoviz src/main.cpp src/file_dialogs.cpp)
    if(RAYLIB_INCLUDE)
        target_include_directories(geoviz PRIVATE "${RAYLIB_INCLUDE}")
    endif()
    target_link_libraries(geoviz PRIVATE geoviz_engine ${GEOVIZ_RAYLIB})
    list(APPEND checked_targets geoviz)
    add_custom_target(run COMMAND geoviz --root "${PROJECT_SOURCE_DIR}" DEPENDS geoviz
        WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    if(WIN32)
        target_link_libraries(geoviz PRIVATE opengl32 gdi32 winmm comdlg32)
        enable_language(RC)
        if(MINGW)
            # Keep windres preprocessing independent of C++ include paths containing spaces.
            add_custom_command(OUTPUT "${PROJECT_BINARY_DIR}/geoviz_icon.obj"
                COMMAND "${CMAKE_RC_COMPILER}" -O coff -i assets/geoviz.rc
                    -o "${PROJECT_BINARY_DIR}/geoviz_icon.obj"
                WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
                DEPENDS assets/geoviz.rc assets/geoviz.ico VERBATIM)
            target_sources(geoviz PRIVATE "${PROJECT_BINARY_DIR}/geoviz_icon.obj")
        else()
            target_sources(geoviz PRIVATE assets/geoviz.rc)
            set_source_files_properties(assets/geoviz.rc PROPERTIES
                INCLUDE_DIRECTORIES "${PROJECT_SOURCE_DIR}"
                OBJECT_DEPENDS "${PROJECT_SOURCE_DIR}/assets/geoviz.ico")
        endif()
    endif()
    install(TARGETS geoviz RUNTIME DESTINATION .)
    install(DIRECTORY assets data docs DESTINATION .)
    install(FILES README.md DESTINATION .)
endif()
foreach(target IN LISTS checked_targets)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(WIN32 AND MINGW AND NOT target STREQUAL "geoviz_engine")
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()
